\begin{frame}{Where We Are}
\textbf{Chapter 5 finding:} scanners disagree on the same image at every level.

\medskip
\begin{itemize}
  \item total counts diverge up to 6.9$\times$
  \item severity agreement on shared CVEs: 7.6--96.3\%
  \item fix-rate divergence: Trivy 77\% vs Grype 34\% on node:14
  \item CRITICAL counts most portable ($\Delta \leq 3$ on 7/9 images)
\end{itemize}

\medskip
\textbf{Conclusion:} raw scanner output is not gate-ready.

\medskip
\textbf{Question this chapter answers:} what \emph{is} gate-ready?
\end{frame}

% ---------------------------------------------------------------
\begin{frame}{Architectural Choice}
Two options considered:

\medskip
\begin{enumerate}
  \item \textbf{Fork HarbourGuard} (existing orchestration platform)
  \item \textbf{Standalone OPA/Rego layer}
\end{enumerate}

\medskip
HarbourGuard rejected on empirical grounds:
\begin{itemize}
  \item fix metadata lost during ingestion (node:14: HG 10\% vs Trivy 77\%)
  \item per-scanner attribution collapsed; cross-scanner consensus broken
\end{itemize}

\medskip
\textbf{Chosen:} standalone OPA layer consuming raw scanner JSON.

Decoupled, deterministic, unit-testable, generalisable.
\end{frame}

% ---------------------------------------------------------------
\begin{frame}[fragile]{Pipeline}
\small
\begin{verbatim}
   trivy / grype / osv-scanner
            |
            v
       normalise.py    (Python adapter per scanner)
            |
            v
     policy_input.json   (unified schema)
            |
            v
       enrich.py       (NVD, OSV, EPSS; cached by CVE)
            |
            v
      classifier       (rule or agent; layer label)
            |
            v
     eol enricher      (endoflife.date / Trivy / agent)
            |
            v
       opa eval        (deterministic policy bundle)
            |
       PASS / REJECT + structured deny report
\end{verbatim}

\medskip
LLMs isolated \emph{outside} the OPA evaluation. Gate decision is deterministic.
\end{frame}

% ---------------------------------------------------------------
\begin{frame}{P1: Any CRITICAL}
\textbf{Rationale.} Baseline. The dumbest gate that still uses the most portable signal.

\medskip
\textbf{Behaviour.} Blocks on any CRITICAL finding from any scanner.

\medskip
\textbf{Results.}
\begin{itemize}
  \item web-dvwa: 328 blocks
  \item juice-shop: 9 blocks
  \item python:3.8: 191 blocks
\end{itemize}

\medskip
\textbf{Failure mode.} Alert fatigue. Blocks indefinitely on unfixable CVEs. Trust erodes; gate gets bypassed.
\end{frame}

% ---------------------------------------------------------------
\begin{frame}{P2: CRITICAL with Available Fix}
\textbf{Rationale.} An unfixable CRITICAL is operationally useless. Block on what developers can act on.

\medskip
\textbf{Behaviour.} P1 conditions + \texttt{fix\_version != null}.

\medskip
\small
\begin{tabular}{lrrr}
\toprule

\midrule

\bottomrule
\end{tabular}

\medskip
\textbf{node:20 case:} 33 CRITICALs, zero have a fix. P2 correctly passes; P1 would block forever.
\end{frame}

\begin{frame}{P4: Enriched CRITICAL}
\textbf{Rationale.} Scanner signals are necessary but insufficient. External data answers \emph{is the CVE real} and \emph{is it being exploited}.

\medskip
Three external signals:
\begin{itemize}
  \item \textbf{NVD status} (Analyzed / Modified; rejected and disputed flags)
  \item \textbf{OSV advisory} (ecosystem confirmation + fix version)
  \item \textbf{EPSS} (exploitation probability, next 30 days)
\end{itemize}

\medskip
Two configurations:

\begin{tabular}{lll}
\toprule
& EPSS & NVD status \\
\midrule
P4 strict & $> 0.1$ & Analyzed only \\
P4 relaxed & $> 0.01$ & Analyzed + Modified \\
\bottomrule
\end{tabular}
\end{frame}

% ---------------------------------------------------------------
\begin{frame}{P5: Layer-Aware Routing}
\textbf{Rationale.} App and OS findings have different prior risk distributions. One threshold cannot serve both.

\medskip
\textbf{Design.}
\begin{itemize}
  \item \textbf{App layer} $\rightarrow$ strict thresholds (EPSS $> 0.1$)
  \item \textbf{OS layer} $\rightarrow$ relaxed thresholds (EPSS $> 0.01$)
\end{itemize}

\medskip
\textbf{Requires.} Each finding labelled \texttt{app | os | unknown} by a classifier.

\medskip
\textbf{Two classifiers evaluated:}
\begin{enumerate}
  \item \textbf{Rule baseline} -- ecosystem string $\to$ layer, deterministic
  \item \textbf{LLM agent} -- semantic judgment, Claude Haiku 4.5
\end{enumerate}
\end{frame}

\begin{frame}{P5 Iteration: NVD ``Modified'' Bug}
\textbf{Initial design.} App-layer required \texttt{NVD status == Analyzed}.

\medskip
\textbf{What happened.} juice-shop passed with 0 blocks. Including CVE-2023-32314 (vm2 sandbox RCE, EPSS 0.7).

\medskip
\textbf{Why.} NVD had reset its status to ``Modified'' (lifecycle update). The strict variant treated this as an invalidity signal.

\medskip
\textbf{Reality.} ``Modified'' is a normal lifecycle state. Only \texttt{Rejected} and \texttt{Disputed} invalidate.

\medskip
\textbf{Fix.} One-line config change. Accept \texttt{[Analyzed, Modified]} for app-layer.

\medskip
\textcolor{accent}{\textbf{Lesson:} parameterised Rego absorbed a policy bug as a config edit. No code change.}
\end{frame}

% ---------------------------------------------------------------
\begin{frame}{P6: EOL-Aware Gating}
\textbf{Rationale.} An end-of-life image will never receive security patches. Further per-CVE triage is operationally pointless.

\medskip
\textbf{Behaviour.}
\begin{itemize}
  \item EOL image $\to$ block immediately, single deny message
  \item Non-EOL image $\to$ delegate to P5\_layer
  \item Configurable: \texttt{eol\_insta\_block: true | false}
\end{itemize}

\medskip
\textbf{Source priority.}
\begin{enumerate}
  \item endoflife.date (canonical per-product lifecycle data)
  \item Trivy \texttt{Metadata.OS.EOSL} (OS distro layer)
  \item LLM fallback (semantic; never invoked on this dataset)
\end{enumerate}
\end{frame}

\begin{frame}{node:20 Lifecycle Transition (Run-time Detection)}
\textbf{At start of study.} node:20 was actively maintained (Group C, modern).

\medskip
\textbf{During the work.} Node.js 20 reached end of maintenance on 2026-04-30.

\medskip
\textbf{What the framework did.} Detected the transition automatically on the next run.

No code change. No dataset update. No hardcoded table to update.

\medskip
\textbf{Why this works.} EOL is queried from endoflife.date at run time, not embedded in code.

\medskip
\textcolor{accent}{\textbf{Takeaway:} policy outcomes evolve with reality when structured external data sources are used.}

A hardcoded EOL table would have continued reporting node:20 as actively maintained indefinitely.
\end{frame}

\begin{frame}{Decision-Quality Audit: Defensible Blocks}
\begin{itemize}
  \item \textbf{node:20: 1 of 33.} Only one has a fix; blocking on the others = infinite build-fail loop with no remediation path.

  \item \textbf{nginx:1.19: 16 of 42.} EOL Debian Buster; the 16 with EPSS $> 0.01$ are the actively-exploited subset.

  \item \textbf{web-dvwa: 116 of 328 (agent).} Strict thresholds on PHP app code; high-EPSS PHP CVEs preserved, low-EPSS filtered.

  \item \textbf{juice-shop: 1 of 9.} CVE-2023-32314 vm2 sandbox escape, EPSS 0.7. Real, known-exploited.
\end{itemize}

\medskip
These are not coincidences. The policy is doing what it was designed to do.
\end{frame}

% ---------------------------------------------------------------
\begin{frame}{Decision-Quality Audit: Contestable Cases}
\begin{itemize}
  \item \textbf{python:3.8: 6 of 191 CRITICALs blocked.}
    \begin{itemize}
      \item 162 of 191 have EPSS $< 0.001$
      \item Defence: EPSS-prioritised gating is supposed to filter these
      \item Counter: 191 CRITICALs is a posture concern regardless of exploitation
      \item \textbf{P6\_strict resolves this} by short-circuiting on image lifecycle
    \end{itemize}

  \medskip
  \item \textbf{juice-shop: 8 of 9 CRITICALs passed P5.}
    \begin{itemize}
      \item All real vulnerabilities; low EPSS = no observed exploitation, not no exploitability
      \item juice-shop is intentionally vulnerable but \emph{not} EOL upstream
      \item P6 does not catch this case
      \item Genuine limitation worth naming
    \end{itemize}
\end{itemize}
\end{frame}

% ---------------------------------------------------------------
\begin{frame}{Limitations (Honestly Named)}
\begin{enumerate}
  \item \textbf{No ground truth for layer classification.} Agent reclassifications look right on inspection; no labelled set exists.

  \medskip
  \item \textbf{Single LLM run per finding.} Stability under multi-run majority voting not measured.

  \medskip
  \item \textbf{Concentrated disagreement on one image.} Layer-aware contribution validated by web-dvwa; generalisation needs more diverse images.

  \medskip
  \item \textbf{No reachability analysis.} The real false-positive problem (vulnerable function exists but is never called) is not addressed. Future direction.

  \medskip
  \item \textbf{Static EPSS snapshot.} Production deployments would re-evaluate periodically.
\end{enumerate}
\end{frame}

% ---------------------------------------------------------------
\begin{frame}{What We Considered and Rejected}
\textbf{P7: AI-based True-Positive / False-Positive classifier.}

\medskip
Why not:
\begin{itemize}
  \item LLM has \emph{less} information than the scanner that did the structured CPE match
  \item Reachability (the real FP problem) needs static analysis, not prose reasoning
  \item Cost of hallucinated suppression is asymmetric (FN ships vulnerability)
  \item No ground truth available for evaluation
  \item Dilutes the architectural claim from ``LLM where structured data fails'' to ``LLM everywhere''
\end{itemize}

\medskip
Full reasoning: \texttt{docs/notes\_considered\_ai\_tp\_classifier.md}
\end{frame}

% ---------------------------------------------------------------
\begin{frame}{Summary}
\begin{itemize}
  \item \textbf{Six-policy progression}, parameterised in Rego, deterministic OPA evaluation.

  \item \textbf{Layer classification} (P5) uses agent only where structured data fails. \textbf{41\% block reduction} on web-dvwa.

  \item \textbf{EOL awareness} (P6) uses endoflife.date primary, Trivy secondary, agent fallback. Caught node:20 transition automatically.

  \item \textbf{Iteration as feature.} NVD-Modified bug surfaced empirically; corrected by config, no code change.

  \item \textbf{Decision-quality audit} is honest about both defensible and contestable outcomes.

  \item \textbf{Framework}, not security oracle. Trade-offs measured, limitations named.
\end{itemize}

\medskip
37/37 OPA tests. $\sim$\$0.30 per full classification run. Reproducible.
\end{frame}
